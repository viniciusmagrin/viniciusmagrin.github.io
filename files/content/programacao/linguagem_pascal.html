Estrutura basica e entrada/saida

O Pascal é fortemente tipado e organizado em blocos explicitados pelas palavras begin e end.


            program EstruturaBasica;

            uses
            SysUtils; // Biblioteca padrão com funções utilitárias

            var
            idade: Integer;
            altura: Real;
            nome: String;

            begin
                Write('Digite nome, idade e altura: ');
                Readln(nome, idade, altura);
                
                // Format para interpolação de strings
                Writeln(Format('Nome: %s | Idade: %d | Altura: %.2f', [nome, idade, altura]));
            end.

Estruturas condicionais


            // 1. if / else
            if nota >= 7.0 then
                begin
                    Writeln('Aprovado');
                end
            else if nota >= 5.0 then
                begin
                    Writeln('Recuperacao');
                end
            else
                begin
                    Writeln('Reprovado'); // Nota: NUNCA usar ponto e vírgula antes do 'else'
                end;

            // 2. case / of (Semelhante ao switch/case, suporta intervalos)
            case opcao of
                1: Writeln('Opcao 1');
                2..5: Writeln('Opcoes de 2 a 5'); // Intervalo
                6, 7: Writeln('Opcao 6 ou 7');    // Lista
            else
                Writeln('Opcao invalida');
            end;

Laços de repetição


            var
            i: Integer;

            begin
            // 1. for (crescente e decrescente)
            for i := 1 to 10 do
                Write(i, ' ');

            for i := 10 downto 1 do
                Write(i, ' ');

            // 2. while (checagem no início)
            i := 1;
            while i <= 10 do
            begin
                Write(i, ' ');
                Inc(i); // Incrementa i em 1 (i := i + 1)
            end;

            // 3. repeat-until (checagem no final, executa pelo menos uma vez)
            repeat
                Write('Digite 0 para sair: ');
                Readln(opcao);
            until opcao = 0;

            // Controle de fluxo: Break (interrompe) e Continue (pula iteração)
            end;

Vetores, Matrizes e Strings


            var
                // Vetores estáticos com limites explícitos
                vetor: array[1..5] of Integer = (10, 20, 30, 40, 50);
                matriz: array[1..2, 1..3] of Integer;
    
                // Array Dinâmico (sem tamanho fixo)
                dinamico: array of Integer;
  
                texto: String;

            begin
                // Redimensionando array dinâmico em tempo de execução
                SetLength(dinamico, 5); // Índices de 0 a 4
                dinamico[0] := 100;

                // Funções de Strings
                texto := 'Linguagem Pascal';
                Writeln(Length(texto));                  // Retorna o tamanho da string
                Writeln(Copy(texto, 1, 9));              // Extrai sub-string ("Linguagem")
                Writeln(UpperCase(texto));               // Converte para maiúsculas
                if SameText(texto, 'LINGUAGEM PASCAL') then // Compara ignorando maiúsculas/minúsculas
                    Writeln('Iguais');
            end;

Ponteiros e Alocação Dinâmica (GetMem, New)

O Pascal oferece suporte completo a ponteiros usando o operador ^ (declaração/desreferenciação) e @ (endereço). 


            type
                PInteger = ^Integer; // Tipo ponteiro para inteiro

            var
                x: Integer;
                p: PInteger;
                ptrDinamico: PInteger;

            begin
                x := 10;
                p := @x; // 'p' recebe o endereço de 'x'

                Writeln('Valor de x: ', x);
                Writeln('Conteudo via p: ', p^); // 'p^' acessa o valor apontado

                // Alocação dinâmica com New e Dispose (para ponteiros tipados)
                New(ptrDinamico);
                ptrDinamico^ := 42;
                
                Dispose(ptrDinamico); // Libera a memória
                ptrDinamico := nil;   // Boa prática (equivalente ao NULL em C)
            end;

Registros (record) e Variantes

Utilizados para agrupar dados heterogêneos.


            type
            TAluno = record
                Nome: String;
                Nota: Real;
            end;

            var
                aluno1: TAluno;
            begin
                aluno1.Nome := 'Carlos';
                aluno1.Nota := 8.5;

                // Bloco 'with' para simplificar o acesso aos campos do registro
                with aluno1 do
                begin
                    Writeln('Aluno: ', Nome);
                    Writeln('Nota: ', Nota:0:1);
                end;
            end;

Manipulação de Arquivos (TextFile e Arquivos Binários)


                var
                    arq: TextFile;
                    linha: String;

                begin
                    // 1. Escrita em Arquivo de Texto
                    AssignFile(arq, 'dados.txt');
                    Rewrite(arq); // Cria ou sobrescreve o arquivo
                    Writeln(arq, 'Texto formatado');
                    CloseFile(arq); // Sempre fechar após usar

                    // 2. Leitura de Arquivo
                    AssignFile(arq, 'dados.txt');
                    Reset(arq); // Abre para leitura
                    
                    while not EOF(arq) do
                    begin
                        Readln(arq, linha);
                        Writeln('Lido: ', linha);
                    end;
                    CloseFile(arq);
                end;

Subprogramas: procedure e function

O Pascal diferencia subprogramas que não retornam valor (procedure) daqueles que retornam (function). Suporta passagem por valor e por referência (usando a palavra-chave var). 


            // Procedimento com passagem por referência (var)
            procedure DobrarValor(var num: Integer);
            begin
                num := num * 2;
            end;

            // Função com retorno (usa a variável oculta 'Result' ou o nome da função)
            function Somar(a, b: Integer): Integer;
            begin
                Result := a + b;
            end;

Recursos Avançados e Orientação a Objetos (Object Pascal)

O Object Pascal traz suporte completo à POO, com classes, herança, encapsulamento, polimorfismo e gerenciamento de exceções.

Declaração de Classe e Métodos


            type
                // Classe base
                TPessoa = class
                private
                    FNome: String; // Atributos privados costumam iniciar com 'F' (Field)
                protected
                    FAttrib: Integer;
                public
                    constructor Create(pNome: String); // Construtor
                    destructor Destroy; override;       // Destrutor
                    
                    procedure Apresentar; virtual;      // Método que pode ser sobrescrito
                    
                    // Encapsulamento via Properties
                    property Nome: String read FNome write FNome;
                end;

                // Herança
                TAlunoOO = class(TPessoa)
                private
                    FNota: Real;
                public
                    constructor Create(pNome: String; pNota: Real);
                    procedure Apresentar; override; // Polimorfismo
                end;

                // Implementação dos Métodos
                constructor TPessoa.Create(pNome: String);
                begin
                    FNome := pNome;
                end;

                destructor TPessoa.Destroy;
                begin
                    inherited Destroy;
                end;

                procedure TPessoa.Apresentar;
                begin
                    Writeln('Pessoa: ', FNome);
                end;

                constructor TAlunoOO.Create(pNome: String; pNota: Real);
                begin
                    inherited Create(pNome); // Chama construtor da classe pai
                    FNota := pNota;
                end;

                procedure TAlunoOO.Apresentar;
                begin
                    Writeln(Format('Aluno: %s | Nota: %.1f', [FNome, FNota]));
                end;

Uso das Classes e Tratamento de Exceções (try..finally / try..except)


            var
                objetoAluno: TAlunoOO;

            begin
                objetoAluno := TAlunoOO.Create('Vinicius', 9.5);
            
                // Bloco try..finally garante a liberação da memória mesmo em caso de erro
                try
                    objetoAluno.Apresentar;
                finally
                    objetoAluno.Free; // 'Free' checa se é nil antes de chamar 'Destroy'
                end;

                // Tratamento de exceções
                try
                    // Algum código suscetível a erros
                except
                    on E: Exception do
                        Writeln('Erro capturado: ', E.Message);
                end;
            end;

Conjuntos (set) e Tipos Enumerados

Recursos nativos muito úteis do Pascal para controle de estados e validações sem correspondente direto em C de forma nativa:


            type
                TDiaSemana = (dom, seg, ter, qua, qui, sex, sab); // Enumerado
                TDias = set of TDiaSemana;                       // Conjunto (Set)

            var
                hoje: TDiaSemana;
                fimDeSemana: TDias;

            begin
                hoje := sab;
                fimDeSemana := [dom, sab];

            // Teste de pertencimento a conjunto
            if hoje in fimDeSemana then
                Writeln('E fim de semana!');
            end;
