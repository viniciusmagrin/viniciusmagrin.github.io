Linguagem C

Estrutura basica e entrada/saida

Para compilar e executar qualquer algoritmo, deve-se usar as bibliotecas padrão e da função principal:


                #include <stdio.h>   // Entrada e saída (printf, scanf, manipulação de arquivos)
                #include <stdlib.h>  // Alocação de memória, conversões, controle de processo
                #include <string.h>  // Manipulação de strings

                int main() {
                    int idade;
                    float altura;
                    char nome[50];

                    // Leitura e exibição
                    printf("Digite nome, idade e altura: ");
                    scanf("%s %d %f", nome, &idade, &altura); // Nota: 'nome' não leva & por ser vetor
                    printf("Nome: %s | Idade: %d | Altura: %.2f\n", nome, idade, altura);

                    return 0; // Indica execução bem-sucedida
                }

Estruturas Condicionais

Utilizadas para desvios condicionais na execução do algoritmo:


                // if / else if / else
                if (nota >= 7.0) {
                    printf("Aprovado\n");
                } else if (nota >= 5.0) {
                    printf("Recuperacao\n");
                } else {
                    printf("Reprovado\n");
                }

                // Operador Ternário (condição ? verdadeiro : falso)
                int maior = (a > b) ? a : b;

                // switch / case (para valores inteiros ou caracteres)
                switch (opcao) {
                    case 1:
                        printf("Opcao 1 selecionada\n");
                        break; // Impede que execute os casos abaixo
                    case 2:
                        printf("Opcao 2 selecionada\n");
                        break;
                    default:
                        printf("Opcao invalida\n");
                }

Laços de Repetição (Loops)

Fundamentais para iteração em vetores, matrizes e buscas:


                // 1. for: ideal quando o número de iterações é conhecido
                for (int i = 0; i < 10; i++) {
                    printf("%d ", i);
                }

                // 2. while: ideal quando a condição depende de uma checagem prévia
                int j = 0;
                while (j < 10) {
                    printf("%d ", j);
                    j++;
                }

                // 3. do-while: garante ao menos uma execução antes de checar a condição
                int opcao;
                do {
                    printf("Digite 0 para sair: ");
                    scanf("%d", &opcao);
                } while (opcao != 0);

                // Controle de fluxo em laços
                // break;    -> interrompe o laço imediatamente
                // continue; -> pula para a próxima iteração do laço

Vetores, Matrizes e Strings

Estruturas de dados homogêneas:


                // Vetor (1D)
                int vetor[5] = {10, 20, 30, 40, 50};

                // Matriz (2D)
                int matriz[2][3] = {
                    {1, 2, 3},
                    {4, 5, 6}
                };

                // Strings (vetor de char terminado em '\0')
                char texto[20] = "Linguagem C";
                int tamanho = strlen(texto);          // Retorna o tamanho da string
                strcpy(texto, "Novo Texto");          // Copia string
                strcmp(texto, "Novo Texto");          // Compara strings (retorna 0 se forem iguais)
                strcat(texto, " Concatenado");        // Concatena strings

Ponteiros e Alocação Dinâmica de Memória

Ponteiros guardam o endereço de memória de outras variáveis:


                int x = 10;
                int *p = &x; // 'p' guarda o endereço de 'x'

                printf("Valor de x: %d\n", x);      // 10
                printf("Endereço de x: %p\n", &x);  // Ex: 0x7ffd...
                printf("Conteúdo via p: %d\n", *p); // 10 (desreferenciação)

                // Alocação Dinâmica (stdlib.h)
                int n = 5;
                int *ptr = (int *) malloc(n * sizeof(int)); // Aloca memória para 5 inteiros

                if (ptr == NULL) {
                    printf("Erro de alocação de memória!\n");
                    exit(1);
                }

                // Uso do ponteiro como array
                for (int i = 0; i < n; i++) {
                    ptr[i] = i * 2;
                }

                // Liberar a memória alocada SEMPRE após o uso
                free(ptr);
                ptr = NULL; // Boa prática para evitar dangling pointer

Estruturas Personalizadas (struct e typedef)

Para criar tipos de dados heterogêneos (essencial para listas encadeadas, árvores, etc.):


                typedef struct {
                char nome[30];
                float nota;
            } Aluno;

            Aluno a1 = {"Carlos", 8.5};

            // Acesso aos campos com variável direta: usa ponto (.)
            printf("Aluno: %s | Nota: %.1f\n", a1.nome, a1.nota);

            // Acesso aos campos via ponteiro: usa seta (->)
            Aluno *ptr_aluno = &a1;
            printf("Aluno via ponteiro: %s\n", ptr_aluno->nome);

Manipulação de Arquivos (FILE *)

Permite ler e escrever dados no disco:


                // 1. Escrita em Arquivo de Texto ("w" cria/sobrescreve, "a" adiciona ao final)
                FILE *arq = fopen("dados.txt", "w");
                if (arq != NULL) {
                    fprintf(arq, "Texto formatado: %d\n", 100);
                    fputs("Linha de texto simples\n", arq);
                    fclose(arq); // Sempre fechar o arquivo
                }

                // 2. Leitura de Arquivo de Texto ("r")
                FILE *arq_leitura = fopen("dados.txt", "r");
                char linha[100];

                if (arq_leitura != NULL) {
                    while (fgets(linha, sizeof(linha), arq_leitura) != NULL) {
                        printf("Lido: %s", linha);
                    }
                    fclose(arq_leitura);
                }

                // 3. Modos Principais de Abertura:
                // "r"  : Leitura
                // "w"  : Escrita (sobrescreve)
                // "a"  : Anexo (append)
                // "rb", "wb" : Leitura/Escrita em modo binário (fread / fwrite)

Funções e Passagem de Parâmetros

Organização em blocos e tipos de passagem:


                // Passagem por valor (cópia)
                void dobrarValor(int num) {
                    num = num * 2;
                }

                // Passagem por referência (usando ponteiros para alterar a variável original)
                void dobrarReferencia(int *num) {
                    *num = (*num) * 2;
                }

                int main() {
                    int valor = 5;
                    
                    dobrarValor(valor);
                    // 'valor' continua sendo 5
                    
                    dobrarReferencia(&valor);
                    // 'valor' agora é 10
                    
                    return 0;
                }
